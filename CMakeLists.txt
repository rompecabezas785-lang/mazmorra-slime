cmake_minimum_required(VERSION 3.16)
project(MazmorraSlime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 2.6.1)
FetchContent_MakeAvailable(SFML)
add_executable(mazmorra_slime src/main.cpp src/Game.cpp)
target_include_directories(mazmorra_slime PRIVATE src)
target_link_libraries(mazmorra_slime PRIVATE sfml-graphics sfml-window sfml-system)
if(MSVC)
  target_compile_options(mazmorra_slime PRIVATE /W4)
else()
  target_compile_options(mazmorra_slime PRIVATE -Wall -Wextra -pedantic)
endif()
